Detalhes:

  • Técnica: Divisão e Conquista
  • Algoritmo: Busca Binária (Busca em vetor ordenado)
  • Autores: John Mauchly (primeira menção formal)
  • Data: 1946
  • Complexidade: O(log n)
  • Semana: Semana 03


#include <stdio.h>
int buscaBinaria(int arr[], int l, int r, int x) {
    while (l <= r) {
        int m = l + (r - l) / 2;
        if (arr[m] == x) return m;
        if (arr[m] < x) l = m + 1;
        else r = m - 1;
    }
    return -1;
}
