Detalhes:

  • Técnica: Iterativo (Incremental)
  • Algoritmo: Inserção Direta (Insertion Sort)
  • Autores: Tradicional
  • Data: Anos 1950
  • Complexidade: O(n^2)
  • Semana: Semana 05


#include <stdio.h>
void insertionSort(int arr[], int n) {
    int i, key, j;
    for (i = 1; i < n; i++) {
        key = arr[i];
        j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j = j - 1;
        }
        arr[j + 1] = key;
    }
}
