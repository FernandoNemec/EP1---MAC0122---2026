Detalhes:

  • Técnica: Iterativo (Força Bruta)
  • Algoritmo: Seleção Direta (Selection Sort)
  • Autores: Tradicional / Vários
  • Data: Anos 1950 (formalização computacional)
  • Complexidade: O(n^2)
  • Semana: Semana 04


#include <stdio.h>
void selectionSort(int arr[], int n) {
    int i, j, min_idx, temp;
    for (i = 0; i < n-1; i++) {
        min_idx = i;
        for (j = i+1; j < n; j++)
            if (arr[j] < arr[min_idx])
                min_idx = j;
        temp = arr[min_idx];
        arr[min_idx] = arr[i];
        arr[i] = temp;
    }
}
