1. Tecnica: tentativa e erro (backtracking)
2. Algoritmo: Problema das N rainhas (N Queens)
3. Autores: Max Bezzel
4. Data: 1848
5. Conceitos: Recursividade, matrizes, tabuleiros, restrições, backtracking
6. Problema: Posicionar N rainhas num tabuleiro de xadrez NxN de tal forma que nenhuma rainha possa atacar outra (não compartilhar a mesma linha, coluna ou diagonal).

7. Complexidade: O(N!)
8. Semana: Semana 04
9. Código:
#include <stdio.h>
#include <stdbool.h>
#define N 4
bool isSafe(int board[N][N], int row, int col) {
    int i, j;
    for (i = 0; i < col; i++)
        if (board[row][i]) return false;
    for (i = row, j = col; i >= 0 && j >= 0; i--, j--)
        if (board[i][j]) return false;
    for (i = row, j = col; j >= 0 && i < N; i++, j--)
        if (board[i][j]) return false;
    return true;
}
bool solveNQUtil(int board[N][N], int col) {
    if (col >= N) return true;
    for (int i = 0; i < N; i++) {
        if (isSafe(board, i, col)) {
            board[i][col] = 1;
            if (solveNQUtil(board, col + 1))
                return true;
            board[i][col] = 0; // backtracking
        }
    }
    return false;
}
